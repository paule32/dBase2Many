# After CDATA.src_file/output_dir are known and before generator.visit(tree):
resource_objects = compile_pascal_resources(

    output_directory=CDATA.output_dir,
    command_line_resources=(
        getattr(args, "resource_files", []) or []
    ),
    include_paths=(
        list(getattr(args, "resource_include_paths", []) or [])
        + list(getattr(args, "includepath", []) or [])
    ),

    CDATA.link_object_files = []

        CDATA.link_object_files.append(filename)
